            const protocol = document.getElementById('cfgProtocol').value;
            const username = document.getElementById('cfgUser').value.trim();
            const password = document.getElementById('cfgPass').value.trim();
            const topicSub = document.getElementById('cfgTopicSub').value.trim();
            const topicPub = document.getElementById('cfgTopicPub').value.trim();

            // Update UI Guide
            document.getElementById('guideTopicSub').innerText = topicSub;
            document.getElementById('guideTopicPub').innerText = topicPub;

            // Disconnect old client if exists
            if (mqttClient) {
                mqttClient.end(true);
            }

            updateStatusBadge('Connecting', 'bg-amber-500', 'Connecting...');
            appendLog('SYSTEM', `Initiating MQTT connection to ${protocol}://${host}:${port}...`);

            const clientId = 'stms_web_' + Math.random().toString(16).substring(2, 10);
            
            const options = {
                clientId: clientId,
                clean: true,
                connectTimeout: 5000,
                reconnectPeriod: 3000
            };

            if (username) options.username = username;
            if (password) options.password = password;

            // Construct WebSocket URL
            // EMQX: wss://broker.emqx.io:8084/mqtt
            // HiveMQ: wss://xxxx.s1.eu.hivemq.cloud:8884/mqtt
            const connectUrl = `${protocol}://${host}:${port}/mqtt`;

            try {
                mqttClient = mqtt.connect(connectUrl, options);

                mqttClient.on('connect', () => {
                    updateStatusBadge('Connected', 'bg-emerald-500', 'Connected');
                    appendLog('SUCCESS', `Connected to MQTT Broker (${host})`);

                    mqttClient.subscribe(topicSub, (err) => {
                        if (!err) {
                            appendLog('SUBSCRIBE', `Subscribed to topic: ${topicSub}`);
                        } else {
                            appendLog('ERROR', `Subscribe error: ${err.message}`);
                        }
                    });
                });

                mqttClient.on('message', (topic, payload) => {
                    handleIncomingMessage(topic, payload.toString());
                });

                mqttClient.on('error', (err) => {
                    updateStatusBadge('Error', 'bg-rose-500', 'Connection Error');
                    appendLog('ERROR', `MQTT Error: ${err.message}`);
                });

                mqttClient.on('close', () => {
                    updateStatusBadge('Disconnected', 'bg-slate-500', 'Disconnected');
                });

            } catch (err) {
                appendLog('ERROR', `Failed to initialize MQTT: ${err.message}`);
            }
        }

        function handleIncomingMessage(topic, rawPayload) {
            appendLog('RX', rawPayload);

            try {
                // Expected JSON Format from ESP32:
                // {"temp": 68.5, "current": 42.0, "dt": 0.45, "faa": 1.25, "fanState": true, "isManual": false, "threshold": 65.0}
                const data = JSON.parse(rawPayload);

                // 1. Update Temperature Metric
                if (data.temp !== undefined) {
                    document.getElementById('valTemp').innerText = Number(data.temp).toFixed(1);
                    const tempBadge = document.getElementById('badgeTempStatus');
                    if (data.temp >= 80) {
                        tempBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-rose-500/20 text-rose-400 border border-rose-500/30 animate-pulse";
                        tempBadge.innerText = "CRITICAL HIGH";
                    } else if (data.temp >= 65) {
                        tempBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-amber-500/20 text-amber-400 border border-amber-500/30";
                        tempBadge.innerText = "WARNING HIGH";
                    } else {
                        tempBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-emerald-500/10 text-emerald-400 border border-emerald-500/20";
                        tempBadge.innerText = "NORMAL";
                    }
                }

                // 2. Update Current & Load %
                if (data.current !== undefined) {
                    document.getElementById('valCurrent').innerText = Number(data.current).toFixed(1);
                    const loadPct = Math.min(100, Math.round((data.current / 100.0) * 100));
                    document.getElementById('valLoadPercent').innerText = loadPct + "%";
                    document.getElementById('barCurrent').style.width = loadPct + "%";
                }

                // 3. Update dT/dt
                if (data.dt !== undefined) {
                    document.getElementById('valDt').innerText = Number(data.dt).toFixed(2);
                    const dtBadge = document.getElementById('badgeDtStatus');
                    if (data.dt >= 3.0) {
                        dtBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-rose-500/20 text-rose-400";
                        dtBadge.innerText = "ANOMALI CEPAT";
                    } else {
                        dtBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-slate-800 text-slate-400";
                        dtBadge.innerText = "STABIL";
                    }
                }

                // 4. Update FAA
                if (data.faa !== undefined) {
                    document.getElementById('valFaa').innerText = Number(data.faa).toFixed(2);
                    const faaBadge = document.getElementById('badgeFaaStatus');
                    if (data.faa > 1.0) {
                        faaBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-amber-500/20 text-amber-400";
                        faaBadge.innerText = "PENUAAN CEPAT";
                    } else {
                        faaBadge.className = "px-2 py-0.5 rounded-full text-[10px] font-bold bg-emerald-500/10 text-emerald-400";
                        faaBadge.innerText = "NORMAL";
                    }
                }

                // 5. Update Fan Status & Mode
                if (data.fanState !== undefined) {
                    const icon = document.getElementById('fanStatusIcon');
                    const text = document.getElementById('fanStatusText');
                    
                    if (data.fanState) {
                        icon.className = "w-12 h-12 rounded-full bg-emerald-500/20 text-emerald-400 flex items-center justify-center text-xl animate-spin";
                        text.innerText = "ON (Berputar)";
                        text.className = "text-lg font-bold text-emerald-400";
                    } else {
                        icon.className = "w-12 h-12 rounded-full bg-slate-800 text-slate-500 flex items-center justify-center text-xl";
                        text.innerText = "OFF (Mati)";
                        text.className = "text-lg font-bold text-slate-400";
                    }
                }

                if (data.isManual !== undefined) {
                    const modeBadge = document.getElementById('fanModeBadge');
                    if (data.isManual) {
                        modeBadge.className = "px-2.5 py-1 rounded-md text-xs font-bold bg-amber-500/10 text-amber-400 border border-amber-500/20";
                        modeBadge.innerText = "MANUAL OVERRIDE";
                    } else {
                        modeBadge.className = "px-2.5 py-1 rounded-md text-xs font-bold bg-blue-500/10 text-blue-400 border border-blue-500/20";
                        modeBadge.innerText = "AUTO";
                    }
                }

                if (data.threshold !== undefined) {
                    document.getElementById('valLimitDisplay').innerText = Number(data.threshold).toFixed(1) + " °C";
                }

                // 6. Update Real-Time Chart Data
                const now = new Date().toLocaleTimeString('id-ID', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' });
                chartLabels.push(now);
                tempDataBuffer.push(data.temp || 0);
                currentDataBuffer.push(data.current || 0);

                if (chartLabels.length > maxPoints) {
                    chartLabels.shift();
                    tempDataBuffer.shift();
                    currentDataBuffer.shift();
                }

                chartInstance.update();

            } catch (e) {
                // Non-JSON plain text message handling
                appendLog('WARN', 'Non-JSON Payload received');
            }
        }

        function sendControl(command) {
            if (!mqttClient || !mqttClient.connected) {
                appendLog('ERROR', 'Cannot send command: MQTT Disconnected');
                return;
            }

            const topicPub = document.getElementById('cfgTopicPub').value.trim();
            const payload = JSON.stringify({ command: command });

            mqttClient.publish(topicPub, payload, { qos: 0 }, (err) => {
                if (!err) {
                    appendLog('TX', `[Topic: ${topicPub}] Payload: ${payload}`);
                } else {
                    appendLog('ERROR', `Publish error: ${err.message}`);
                }
            });
        }

        function sendLimit() {
            const limitVal = parseFloat(document.getElementById('inputLimit').value);
            if (isNaN(limitVal) || limitVal < 40 || limitVal > 90) {
                appendLog('WARN', 'Limit must be between 40 and 90 °C');
                return;
            }

            if (!mqttClient || !mqttClient.connected) {
                appendLog('ERROR', 'Cannot send limit: MQTT Disconnected');
                return;
            }

            const topicPub = document.getElementById('cfgTopicPub').value.trim();
            const payload = JSON.stringify({ command: 'SET_LIMIT', value: limitVal });

            mqttClient.publish(topicPub, payload, { qos: 0 }, (err) => {
                if (!err) {
                    appendLog('TX', `[Set Limit] Payload: ${payload}`);
                } else {
                    appendLog('ERROR', `Publish error: ${err.message}`);
                }
            });
        }

        function updateStatusBadge(statusText, dotColorClass, textLabel) {
            const statusDot = document.getElementById('statusDot');
            const statusTextEl = document.getElementById('statusText');
            statusDot.className = `w-2.5 h-2.5 rounded-full ${dotColorClass} ${statusText === 'Connecting' ? 'animate-ping' : ''}`;
            statusTextEl.innerText = textLabel;
        }

        function appendLog(type, msg) {
            const container = document.getElementById('terminalLog');
            const logItem = document.createElement('div');
            const timeStr = new Date().toLocaleTimeString('id-ID', { hour12: false });
            
            let typeColor = 'text-slate-400';
            if (type === 'RX') typeColor = 'text-emerald-400 font-semibold';
            if (type === 'TX') typeColor = 'text-amber-400 font-semibold';
            if (type === 'ERROR') typeColor = 'text-rose-400 font-bold';
            if (type === 'SUCCESS') typeColor = 'text-sky-400';

            logItem.innerHTML = `<span class="text-slate-600">[${timeStr}]</span> <span class="${typeColor}">[${type}]</span> ${escapeHtml(msg)}`;
            container.appendChild(logItem);
            container.scrollTop = container.scrollHeight;

            logCounter++;
            document.getElementById('logCount').innerText = `${logCounter} logs`;
        }

        function clearLogs() {
            document.getElementById('terminalLog').innerHTML = '<div class="text-slate-500">[SYSTEM] Console logs cleared.</div>';
            logCounter = 0;
            document.getElementById('logCount').innerText = '0 logs';
        }

                document.getElementById('cfgUser').value = 'your_username';
                document.getElementById('cfgPass').value = 'your_password';
            }
        }

        function escapeHtml(text) {
            return text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
